""" GitHub App authentication: private-key JWT → installation access token. Why prefer this over a PAT in production: - Tokens are short-lived (1 hour); no long-lived user secrets on your server. - Scoped per-installation (just this repo, just the permissions you grant). - Survives the original user leaving the org. Setup: 1. Create a GitHub App (Developer Settings → GitHub Apps → New). Permissions: Repository contents = Read-only. Metadata = Read-only. 2. Generate a private key (.pem) and store the path in GITHUB_APP_PRIVATE_KEY_PATH. 3. Install the app on your vault repo. Copy the installation ID from the URL. 4. Set env: GITHUB_APP_ID, GITHUB_INSTALLATION_ID, GITHUB_APP_PRIVATE_KEY_PATH. Deps: pip install pyjwt cryptography requests """ import os import time from datetime import datetime import jwt import requests APP_ID = os.environ.get("GITHUB_APP_ID") INSTALLATION_ID = os.environ.get("GITHUB_INSTALLATION_ID") PRIVATE_KEY_PATH = os.environ.get("GITHUB_APP_PRIVATE_KEY_PATH") _cache = {"token": None, "expires_at": 0.0} def _app_jwt() -> str: """Short-lived JWT signed with the app's private key. Identifies the app to GitHub.""" with open(PRIVATE_KEY_PATH, "rb") as f: private_key = f.read() now = int(time.time()) payload = { "iat": now - 60, # backdate 60s to tolerate clock drift "exp": now + 9 * 60, # max allowed is 10 min "iss": APP_ID, } return jwt.encode(payload, private_key, algorithm="RS256")def get_installation_token() -> str: """Exchange the app JWT for a 1-hour installation access token. Cached & auto-refreshed." if _cache["token"] and _cache["expires_at"] - 300 > time.time(): return _cache["token"] r = requests.post( f"https://api.github.com/app/installations/{INSTALLATION_ID}/access_tokens", headers={ "Authorization": f"Bearer {_app_jwt()}", "Accept": "application/vnd.github+json", "X-GitHub-Api-Version": "2022-11-28", }, timeout=10, ) r.raise_for_status() data = r.json() _cache["token"] = data["token"] _cache["expires_at"] = datetime.fromisoformat( data["expires_at"].replace("Z", "+00:00") ).timestamp() return _cache["token"]
